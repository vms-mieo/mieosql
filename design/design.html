-- Query top selling products with inventory count
SELECT
p.ProductID,
p.Name AS ProductName,
p.ProductNumber,
p.Color,
p.ListPrice,
COALESCE(SUM(inv.Quantity), 0) AS TotalStock,
p.ModifiedDate
FROM Production.Product p
LEFT JOIN Production.ProductInventory inv ON p.ProductID = inv.ProductID
WHERE p.ListPrice > 50.00 AND p.SellEndDate IS NULL
GROUP BY p.ProductID, p.Name, p.ProductNumber, p.Color, p.ListPrice, p.ModifiedDate
ORDER BY p.ListPrice DESC LIMIT 100;
